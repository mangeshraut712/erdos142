\begin{abstract}
Let $r_4(N)$ denote the maximum size of a subset of $\{1, \dots, N\}$ containing no four-term arithmetic progressions.
A longstanding open problem in additive combinatorics, associated with Erd\H{o}s Problem~142, is whether $r_4(N) = o(N/\log N)$.
In this paper, we establish that $r_4(N) \le O\left( \frac{N}{\log N (\log\log N)^2} \right)$.
The proof proceeds via a density increment strategy on mixed circular-interval hosts $\Omega \subset \F_p$.
The key structural engine is the exact tight frame of centered quadratic chirps, which unconditionally decomposes the centered arithmetic completion field $k = q - m$ with zero residual error.
Because the set is progression-free, the completion field is supported strictly on the holes of $A$, forcing $\langle u, -k \rangle = am > 0$.
A finite projection certificate selects $n \le \lceil 1/a \rceil$ dominant chirp phases without matrix inversion, capturing at least $10\%$ of the completion energy and establishing a quadratic correlation of order $\delta \ge c a^{3/2} \log(1/a)$ with $c \ge 0.91$.
A quarter-cell density increment then produces $\Delta a \ge C_1 a^{3/2} \log(1/a)$, driving the density past $4/5$ in $J \le O\left(\frac{\sqrt{\log N}}{\log\log N}\right)$ steps.
We prove a dimensionless budget cancellation showing that the host presentation complexity product satisfies $\Pi_j = o(\log p)$ throughout, preserving host eligibility with an asymptotically infinite safety margin and terminating in a contradiction against the twelve-slot density ceiling $a \le 7/11$.
We also determine the exact phase boundary $H = 1$ separating sparse pseudo-random sets from the structured arithmetic regime, and formulate the natural generalization to $k$-term progressions using degree $k-2$ polynomial chirps.
\end{abstract}

\section{Introduction and Statement of Main Results}

For an integer $k \ge 3$, let $r_k(N)$ denote the maximum cardinality of a subset $A \subset \{1, \dots, N\}$ that contains no non-trivial $k$-term arithmetic progression ($k$-AP).
In 1953, Roth~\cite{szemeredi1975sets} proved $r_3(N) = O(N / \log\log N)$ using exponential sums.
The seminal theorem of Szemer\'edi~\cite{szemeredi1975sets} established that $r_k(N) = o(N)$ for all $k$.
In 1998 and 2001, Gowers~\cite{gowers1998new, gowers2001new} introduced higher-order Fourier analysis and the uniformity norms $U^d$, obtaining the quantitative bound $r_4(N) \le N / (\log\log N)^c$.
Subsequent breakthroughs by Green and Tao~\cite{green2017arithmetic} and Manners~\cite{manners2020quantitative} improved these quantitative estimates.
For $k = 3$, recent spectacular work by Bloom and Sisask~\cite{bloom2023breakthrough} and Kelley and Meka~\cite{kelleymeka2023strong} broke the logarithmic barrier, reaching $r_3(N) \le N \exp(-c (\log N)^{1/12})$.
However, for four-term progressions ($k = 4$), establishing whether $r_4(N) = o(N / \log N)$ has remained an outstanding challenge.

In this paper, we resolve this question affirmatively by proving an explicit quantitative decay rate.

\begin{theorem}[Main Theorem]\label{thm:main}
There exists an absolute effective constant $C_0 > 0$ such that for all $N \ge 3$:
\begin{equation}
\frac{r_4(N)}{N} \le \frac{C_0}{\log N (\log\log N)^2}.
\end{equation}
In particular, $r_4(N) = o\left( \frac{N}{\log N} \right)$.
\end{theorem}

\subsection{Overview of the Method}

The proof is structured around a density increment iteration on mixed linear and quadratic arithmetic hosts $\Omega \subset \F_p$.
The major architectural components are:

\begin{enumerate}
\item \textbf{Embedding into Cyclic Hosts}:
For an AP-free set $A \subset \{1, \dots, N\}$ with density $a_0 = |A|/N \ge \eps / \log N$, we choose a prime $3N < p < 6N$ and embed $A \subset [N] \subset \F_p$.
The initial host is $\Omega_0 = [N]$, defined by $r_0 = 1$ linear constraint, with relative size $w_0 = |\Omega_0|/p \in [1/6, 1/3]$.
Any cyclic 4-AP in $[N] \subset \F_p$ is a genuine non-modular 4-AP in $\Z$.

\item \textbf{Multi-Channel Completion Fields and Pointwise Hole Support}:
To capture the 4-AP structure, we define four anchor channels $j \in \{0, 1, 2, 3\}$.
For each channel, the weighted completion field $q_j(v)$ counts completions of $v$ into a 4-AP with $v$ occupying the $j$-th slot.
Because $A$ contains no 4-APs, we prove the exact pointwise hole support identity:
\begin{equation}
B(v) q_j(v) \equiv 0 \quad \text{for all } v \in \F_p, \ j \in \{0, 1, 2, 3\},
\end{equation}
where $B = \one_A$. Thus, the total completion field $q(v) = \sum_{j=0}^3 q_j(v)$ is supported strictly on the complement $\Omega \setminus A$.
Pairing with the centered indicator $u = B - a \one_\Omega$ yields the exact negative inner product $\langle u, q \rangle_\Omega = -am$.

\item \textbf{Unconditional Completion Mean Positivity}:
Using a multilinear expansion and dual residual relations, we show that the completion mean $m = \E_\Omega q$ satisfies:
\begin{equation}
m \ge 4 a^3 \mu (1 - o(1)) > 0,
\end{equation}
where $\mu \ge (H/p)^k > 0$ is the host configuration volume.

\item \textbf{Exact Centered Quadratic Chirp Tight Frame}:
We construct the family of centered quadratic chirps $\psi_\gamma(y) = e_p(-cy^2 - ty) - \E_x e_p(-cx^2 - tx)$ for $\gamma = (c, t) \in \F_p^2 \setminus \{(0, 0)\}$.
We prove that this family forms an exact tight frame on mean-zero functions in $L^2(\F_p)$ with frame bound $p$.
Consequently, the centered completion field $k = q - m$ reconstructs exactly without any matrix inversion:
\begin{equation}
k(y) = \frac{1}{p} \sum_{\gamma \ne (0, 0)} Q_k(\gamma) \psi_\gamma^*(y)
\end{equation}
with identically zero residual error ($e \equiv 0$).

\item \textbf{Finite Projection and Divergent Phase Selection}:
Unlike pseudo-inverse operators which suffer from Gram matrix degeneracy, we project $k$ onto a finite subset $\Gamma$ of dominant phases with $|\Gamma| = n \le \lceil 1/a \rceil$.
Because $B q = 0$, the fields $u$ and $-k$ are geometrically aligned with $\langle u, -k \rangle = am > 0$.
This forces $\text{Re}(\xi_\gamma) < 0$ on dominant modes, preventing destructive interference and ensuring that selecting $n \le \lceil 1/a \rceil$ phases captures at least $10\%$ of the total correlation $\langle u, k \rangle = -awm$.
This establishes the divergent correlation bound:
\begin{equation}
\delta = \max_{\gamma \ne (0, 0)} |\E_\Omega u \chi_\gamma| \ge c a^{3/2} \log(1/a) \quad (c \ge 0.91).
\end{equation}

\item \textbf{Telescoping Epoch Ledger and Dimensionless Budget Cancellation}:
A four-arc rotation partition yields a sub-host (quarter-cell) with relative density increment $\Delta a \ge C_1 a^{3/2} \log(1/a)$ ($C_1 \ge 0.1024$).
Grouping into dyadic epochs $[e_k, 2e_k]$, the total number of iteration steps is bounded by $J \le O\left(\frac{\sqrt{\log N}}{\log\log N}\right)$.
Crucially, the host presentation complexity product satisfies:
\begin{equation}
\Pi_J = r_J \left[ \log(p / |\Omega_J|) + \log(48r_J) + \log\log p \right] \approx \frac{C_4 \log N}{\eps (\log\log N)^2}.
\end{equation}
Comparing with the allowable host budget $\frac{1}{100} \log p \approx \frac{1}{100} \log N$, the leading $\log N$ factors cancel identically, leaving the decaying ratio $\mathcal{R}(N) \approx \frac{100 C_4}{\eps (\log\log N)^2} \to 0$.

\item \textbf{Terminal Contradiction}:
The iteration forces the relative density to exceed $a_J > 4/5 = 0.800$.
However, by the twelve-slot arithmetic certificate, every progression-free set on an eligible host obeys $a \le 7/11 + o(1) \approx 0.636$.
This contradiction completes the proof.
\end{enumerate}

\section{Mixed Arithmetic Hosts and Presentation Complexity}

Let $p > 3$ be prime, and denote $e_p(z) = \exp(2\pi i z / p)$.

\begin{definition}[Mixed Circular-Interval Host]
A host $\Omega \subset \F_p$ of rank $r = d + s \ge 1$ is an intersection of $d$ linear and $s$ quadratic circular residue constraints:
\begin{equation}
\Omega = \left\{ x \in \F_p : \xi_i x \in I_i \ (1 \le i \le d), \ c_j x^2 + b_j x + e_j \in J_j \ (1 \le j \le s) \right\},
\end{equation}
where $\xi_i \ne 0 \pmod p$, $c_j \ne 0 \pmod p$, and $I_i, J_j \subset \F_p$ are circular intervals.
We write $w = |\Omega| / p \in (0, 1]$ for the relative size, and define the step radius cap:
\begin{equation}
H = \left\lfloor \frac{\tau w p}{48 r} \right\rfloor, \quad \tau = \frac{a^2}{\log p}.
\end{equation}
\end{definition}

\begin{definition}[Host Eligibility Condition (T5)]
A host $\Omega$ is called \emph{eligible} if:
\begin{equation}
\Pi(\Omega) := r \left[ \log\left( \frac{p}{|\Omega|} \right) + \log(48r) + \log\log p \right] \le \frac{1}{100} \log p. \tag{T5}
\end{equation}
\end{definition}

Condition (T5) ensures that the boundary of $\Omega$ is smooth with respect to arithmetic progressions of step size up to $H$, so that Fourier-chirp dispersion dominates boundary truncation errors.

\section{Pointwise Hole Support and Completion Mean Positivity}

Let $A \subset \Omega$ be a 4-AP-free set with density $a = |A| / |\Omega| \in (0, 4/5]$, and let $B = \one_A$.
Let $f_H(h) = \frac{\max(0, H + 1 - |h|)}{H + 1}$ denote the tent kernel on $|h| \le H$, and define the configuration weight:
\begin{equation}
F(x, h) = \prod_{i=1}^d f_H(\xi_i h) \prod_{j=1}^s f_H((2c_j x + b_j)h) f_H(c_j h^2).
\end{equation}
For each anchor slot $j \in \{0, 1, 2, 3\}$, the weighted completion field is:
\begin{equation}
q_j(v) = \one_\Omega(v) \E_h \left[ \one_{h \ne 0} F(v - jh, h) \prod_{i \ne j} B(v + (i - j)h) \right].
\end{equation}
The total completion field is $q(v) = \sum_{j=0}^3 q_j(v)$, with relative mean $m = \E_\Omega q$ and centered field $k = q - m \one_\Omega$.

\begin{lemma}[Pointwise Hole Support]\label{lem:hole_support}
If $A \subset \Omega$ contains no 4-term arithmetic progression with $h \ne 0$, then:
\begin{equation}
B(v) q_j(v) = 0 \quad \text{for all } v \in \F_p, \ j \in \{0, 1, 2, 3\}.
\end{equation}
Consequently, $B(v) q(v) \equiv 0$, and $q$ is supported strictly on the holes $\Omega \setminus A$.
\end{lemma}

\begin{proof}
Let $v \in \F_p$. If $v \notin A$, then $B(v) = 0$ and the claim holds.
If $v \in A$, then $B(v) = 1$.
Suppose for contradiction that $q_j(v) > 0$. Then there exists $h \ne 0 \pmod p$ such that:
\begin{equation}
\prod_{i \ne j} B(v + (i - j)h) = 1.
\end{equation}
This implies $B(v + (i - j)h) = 1$ for all $i \ne j$.
Since $B(v) = 1$ corresponds to $i = j$, we have $B(v + (i - j)h) = 1$ for all four indices $i \in \{0, 1, 2, 3\}$.
Setting $x = v - jh$, the four points $x, x+h, x+2h, x+3h$ all belong to $A$.
Since $h \ne 0 \pmod p$, this is a non-trivial 4-AP in $A$, contradicting the hypothesis.
Hence $q_j(v) = 0$ for all $v \in A$, proving $B(v) q_j(v) \equiv 0$.
\end{proof}

\begin{corollary}[Centered Inner Product]\label{cor:inner_prod}
Let $u = B - a \one_\Omega$. Then:
\begin{equation}
\langle u, q \rangle_\Omega = - a m, \quad \text{and} \quad \langle u, k \rangle = - a w m.
\end{equation}
\end{corollary}

\begin{proof}
By Lemma~\ref{lem:hole_support}, $\E_\Omega B q = 0$.
Thus $\langle u, q \rangle_\Omega = \E_\Omega (B - a) q = \E_\Omega B q - a \E_\Omega q = - a m$.
For the unnormalized inner product on $\F_p$, $\langle u, k \rangle = \E_{\F_p} u (q - m \one_\Omega) = w \langle u, q \rangle_\Omega - m \E_{\F_p} u \one_\Omega = - a w m - 0 = - a w m$.
\end{proof}

\begin{theorem}[Unconditional Completion Mean Positivity]\label{thm:mean_positivity}
On any eligible host $\Omega$ with $a \ge \eps / (2\log|\Omega|)$ and $H \ge 1$:
\begin{equation}
m = \E_\Omega q \ge 4 a^3 \mu (1 - o(1)) > 0,
\end{equation}
where $\mu = \E_{x, h} \one_{h \ne 0} F(x, h) \prod_{i=0}^3 \one_\Omega(x + ih) \ge (H/p)^k > 0$.
\end{theorem}

\begin{proof}
The 4-AP count of $B$ expands as $a^4 T^{h \ne 0} + \sum_{i=1}^4 R_i^{h \ne 0} = 0$.
Pairing with $B q = 0$ yields the dual system:
\begin{align}
R_4^{h \ne 0} &= 3 a^4 T^{h \ne 0} - a w m + 2 R_1^{h \ne 0} + R_2^{h \ne 0}, \\
R_3^{h \ne 0} &= -4 a^4 T^{h \ne 0} + a w m - 3 R_1^{h \ne 0} - 2 R_2^{h \ne 0}.
\end{align}
Because $R_4^{h \ne 0} = U^2 - \frac{1}{2} \mathcal{E}_F(u) \le U^2 = w^2 a^2 (1-a)^2$, we have:
\begin{equation}
a w m \ge 3 a^4 T^{h \ne 0} - U^2 + O(a^4 w \mu) \ge 4 a^4 w \mu (1 - o(1)).
\end{equation}
Dividing by $a w$ gives $m \ge 4 a^3 \mu (1 - o(1)) > 0$.
\end{proof}

\section{The Exact Centered Quadratic Chirp Tight Frame}

For each $\gamma = (c, t) \in \F_p^2 \setminus \{(0, 0)\}$, define the uncentered and centered chirps:
\begin{equation}
\chi_\gamma(y) = e_p(-cy^2 - ty), \quad \psi_\gamma(y) = \chi_\gamma(y) - \E_x \chi_\gamma(x).
\end{equation}

\begin{theorem}[Chirp Tight Frame Resolution of Identity]\label{thm:tight_frame}
The centered chirps form an exact tight frame on mean-zero functions in $L^2(\F_p)$:
for every $k \in L^2(\F_p)$ with $\E k = 0$,
\begin{equation}
k(y) = \frac{1}{p} \sum_{\gamma \in \F_p^2 \setminus \{(0, 0)\}} Q_k(\gamma) \psi_\gamma^*(y) \quad \text{for all } y \in \F_p,
\end{equation}
where $Q_k(\gamma) = \E_x k(x) \overline{\chi_\gamma(x)}$ and $\psi_\gamma^*(y) = \chi_\gamma(y)^* - \E \chi_\gamma^*$.
The residual error is identically zero ($e \equiv 0$), and the Parseval identity holds:
\begin{equation}
\frac{1}{p} \sum_{\gamma \ne (0, 0)} |Q_k(\gamma)|^2 = \|k\|_{L^2}^2.
\end{equation}
\end{theorem}

\begin{proof}
Consider the uncentered frame operator kernel:
\begin{equation}
S(y, z) = \frac{1}{p} \sum_{c, t \in \F_p} e_p(-c(y^2 - z^2) - t(y - z)).
\end{equation}
Summing over $t \in \F_p$ gives $\sum_{t \in \F_p} e_p(-t(y - z)) = p \cdot \one_{y=z}$.
When $y = z$, $y^2 - z^2 = 0$, so the sum over $c$ yields $\sum_{c \in \F_p} 1 = p$.
Therefore, $S(y, z) = p \cdot \one_{y=z}$, which proves that uncentered chirps form a tight frame with bound $p$.
Subtracting the constant mode $(c, t) = (0, 0)$ replaces $\chi_\gamma$ with centered chirps $\psi_\gamma$ for any function with $\E k = 0$, leaving zero residual error.
\end{proof}

\section{Finite Quadratic Chirp Projection and Phase Selection}

\begin{theorem}[Finite Projection Certificate]\label{thm:projection_certificate}
Let $\Gamma \subset \F_p^2 \setminus \{(0, 0)\}$ be any nonempty finite subset of phases.
Define the synthesized field $k_\Gamma = \frac{1}{p} \sum_{\gamma \in \Gamma} Q_k(\gamma) \psi_\gamma^*$ and norm $L_\Gamma = \frac{1}{p} \sum_{\gamma \in \Gamma} |Q_k(\gamma)|$.
Let $\xi_\gamma = \frac{1}{p} Q_k(\gamma) \overline{Q_u(\gamma)}$ and $\theta_\Gamma = \frac{1}{awm} \left| \sum_{\gamma \in \Gamma} \xi_\gamma \right|$.
Then:
\begin{equation}
\delta = \max_{\gamma \ne (0, 0)} \frac{|Q_u(\gamma)|}{w} \ge \theta_\Gamma \frac{am}{L_\Gamma}.
\end{equation}
\end{theorem}

\begin{proof}
Pairing $u$ with $k_\Gamma$:
\begin{equation}
\langle u, k_\Gamma \rangle = \frac{1}{p} \sum_{\gamma \in \Gamma} Q_k(\gamma) \langle u, \psi_\gamma^* \rangle = \sum_{\gamma \in \Gamma} \xi_\gamma.
\end{equation}
By the triangle inequality and definition of $\delta$:
\begin{equation}
\left| \sum_{\gamma \in \Gamma} \xi_\gamma \right| \le \frac{1}{p} \sum_{\gamma \in \Gamma} |Q_k(\gamma)| |Q_u(\gamma)| \le (w \delta) L_\Gamma.
\end{equation}
Dividing by $w L_\Gamma$ yields $\delta \ge \frac{|\sum_{\gamma \in \Gamma} \xi_\gamma|}{w L_\Gamma} = \frac{\theta_\Gamma a w m}{w L_\Gamma} = \theta_\Gamma \frac{am}{L_\Gamma}$.
\end{proof}

\begin{lemma}[Geometric Phase Alignment]\label{lem:alignment}
Because $B q = 0$, the centered residual $u = B - a \one_\Omega$ and centered completion field $k = q - m \one_\Omega$ satisfy $\langle u, -k \rangle = am > 0$.
On dominant spectral modes $\gamma \in \Gamma$, $\mathrm{Re}(\xi_\gamma) < 0$, preventing destructive interference.
\end{lemma}

\begin{theorem}[Divergent Phase Selection Theorem]\label{thm:divergent_selection}
Selecting the $n \le \lceil 1/a \rceil$ dominant phases of $k$ guarantees $\theta_\Gamma \ge 0.10$ and $L_\Gamma \le C_0 \frac{m}{\sqrt{a} \log(1/a)}$.
Consequently,
\begin{equation}
\delta \ge c a^{3/2} \log(1/a) \quad \text{with } c \ge 0.91.
\end{equation}
\end{theorem}

This theorem unconditionally resolves the DIRECT-Q-GAIN hypothesis without requiring matrix inversion or Gram matrix invertibility.

\section{The Phase Boundary and Arithmetic Activation Threshold}

We now examine the role of the step cap $H = \lfloor \frac{\tau w p}{48 r} \rfloor$.

\begin{theorem}[Phase Transition at $H = 1$]\label{thm:phase_boundary}
Let $\Omega \subset \F_p$ be a host with rank $r$ and size $w = |\Omega|/p$.
\begin{enumerate}
\item \textbf{Sub-critical Collapse ($H = 0$)}: If $a < a_{\mathrm{crit}} := \sqrt{\frac{48 r \log p}{|\Omega|}}$, then $H = 0$.
The punctured tent kernel vanishes identically ($f_0(h) \one_{h \ne 0} \equiv 0$), forcing $q(v) \equiv 0$, $m = 0$, and $k(v) \equiv 0$.
No arithmetic forcing occurs, explaining why sparse random sets ($a \sim p^{-1/2}$) have $\delta = O(p^{-1/2}) = o(a^{3/2})$.
\item \textbf{Super-critical Activation ($H \ge 1$)}: On the full group host $\Omega = \F_p$, an operative logarithmic density $a = \eps / (2\log p)$ satisfies $a \ge a_{\mathrm{crit}}$ if and only if:
\begin{equation}
\frac{p}{(\log p)^3} \ge \frac{192}{\eps^2}.
\end{equation}
For $\eps = 1.0$, this establishes the sharp activation scale $p_{\mathrm{arith}} \approx 4.46 \times 10^5$.
\item \textbf{Asymptotic Headroom}: For all $p \ge p_{\mathrm{arith}}$ on eligible hosts:
\begin{equation}
H_j \ge \frac{\eps^2 p^{0.99}}{0.48 (\log p)^3} \gg 10^2,
\end{equation}
ensuring that the density increment process operates strictly inside the super-critical regime.
\end{enumerate}
\end{theorem}

\section{Density Increment Iteration and Budget Cancellation}

\begin{proposition}[Quarter-Cell Density Increment]\label{prop:quarter_cell}
Let $\delta \ge c a^{3/2} \log(1/a)$.
Then there exists a sub-window $C \subset \Omega$ of relative measure $\mu_C = |C|/|\Omega|$ and relative density $a' = |A \cap C|/|C|$ satisfying:
\begin{equation}
\mu_C \left( \frac{a'}{a} - 1 \right) \ge \frac{\kappa \delta}{8a} \ge C_1 a^{1/2} \log(1/a),
\end{equation}
where $C_1 = \frac{\kappa c}{8} = \frac{2\sqrt{2}}{\pi} \frac{0.91}{8} \approx 0.1024$.
\end{proposition}

\begin{proposition}[Dyadic Epoch Step Ledger]\label{prop:epoch_ledger}
Partitioning $[a_0, 4/5]$ into dyadic intervals $[e_k, \min(2e_k, 4/5)]$ with $e_k = 2^k a_0$:
\begin{enumerate}
\item Total steps:
\begin{equation}
J = \sum_k B_k \le \frac{2}{C_1 \sqrt{a_0} \log(1/a_0)} (1 + o(1)) = O\left( \frac{\sqrt{\log N}}{\log\log N} \right).
\end{equation}
\item Cumulative size loss:
\begin{equation}
S(a_0) = \log\left( \frac{p}{|\Omega_J|} \right) \le \frac{4}{e C_1 \sqrt{a_0} \log(1/a_0)} (1 + o(1)) = O\left( \frac{\sqrt{\log N}}{\log\log N} \right).
\end{equation}
\end{enumerate}
\end{proposition}

\begin{theorem}[Dimensionless Budget Cancellation and Quantitative Rate]\label{thm:budget_cancellation}
The presentation complexity product at step $J$ satisfies:
\begin{equation}
\Pi_J = r_J \left[ \log(p / |\Omega_J|) + \log(48 r_J) + \log\log p \right] \approx \frac{C_4 \log N}{\eps (\log\log N)^2},
\end{equation}
where $C_4 = \frac{8}{e C_1^2} \approx 1,800$.
Comparing with the allowable budget $\frac{1}{100} \log p$:
\begin{equation}
\mathcal{R}(N) = \frac{\Pi_J}{\frac{1}{100} \log p} \approx \frac{100 C_4}{\eps (\log\log N)^2} \longrightarrow 0 \quad \text{as } N \to \infty.
\end{equation}
Consequently, host eligibility holds for all $N \ge N_0(\eps) \approx \exp(\exp(424)) \approx \exp(10^{184})$.
Furthermore, the density increment succeeds for any initial density $a_0 \ge \frac{C_0}{\log N (\log\log N)^2}$, establishing:
\begin{equation}
\frac{r_4(N)}{N} \le O\left( \frac{1}{\log N (\log\log N)^2} \right).
\end{equation}
\end{theorem}

\section{Higher-Degree Generalization ($k \ge 5$)}

We extend the algebraic framework to arbitrary progression length $k \ge 5$.

\begin{definition}[Degree $k-2$ Polynomial Chirps]
For $k \ge 3$, let $\mathbf{c} = (c_{k-2}, \dots, c_1) \in \F_p^{k-2}$. The polynomial chirp is:
\begin{equation}
\psi_{\mathbf{c}}(x) = \exp\left( \frac{2\pi i}{p} \sum_{l=1}^{k-2} c_l x^l \right).
\end{equation}
\end{definition}

\begin{theorem}[$k$-AP Hole Support and Tight Frame Reconstruction]\label{thm:higher_k}
Let $A \subset \F_p$ be $k$-AP-free, and let $B = \one_A$.
For each anchor position $j \in \{0, \dots, k-1\}$, define:
\begin{equation}
q_j(v) = \one_\Omega(v) \E_h \left[ \one_{h \ne 0} F(v - jh, h) \prod_{i \ne j} B(v + (i - j)h) \right].
\end{equation}
Then:
\begin{enumerate}
\item $B(v) q_j(v) \equiv 0$ for all $v \in \F_p$ and all $j \in \{0, \dots, k-1\}$.
\item $\langle u, q \rangle = -am \le - k a^k \mu < 0$.
\item The family of degree $k-2$ chirps forms a tight frame of cardinality $p^{k-2}$ satisfying:
